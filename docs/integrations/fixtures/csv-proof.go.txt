package main

import (
	"encoding/csv"
	"os"

	"github.com/lets-parley/parley/internal/poker"
	"github.com/lets-parley/parley/internal/session"
)

func main() {
	half, ordinal, zero := "½", "M", "0"
	rows, err := poker.Kind().CSV(session.Envelope{Kind: "poker", State: poker.State{Stories: []poker.WireStory{
		{Ref: "EXAMPLE-1", Title: "Parent history", Status: "done", Estimate: &half},
		{Ref: "", Title: "Independent candidate, unestimated", Status: "pending"},
		{Ref: "", Title: "Ordinal candidate", Status: "done", Estimate: &ordinal},
		{Ref: "", Title: "Explicit zero", Status: "done", Estimate: &zero},
		{Ref: "", Title: "=UNTRUSTED()", Status: "pending"},
	}}})
	if err != nil {
		panic(err)
	}
	if rows[1][3] != "½" || rows[2][3] != "" || rows[3][3] != "M" || rows[4][3] != "0" || rows[5][1] != "'=UNTRUSTED()" {
		panic("CSV labels, blank, zero, or formula escaping changed")
	}
	w := csv.NewWriter(os.Stdout)
	if err := w.WriteAll(rows); err != nil {
		panic(err)
	}
}
